{{- /* Resolves a URL for output in href/src. Site-root paths ("/book-a-call/") are rewritten relative to baseURL so they keep working when the site is served from a subdirectory. Note the leading slash must be trimmed first: Hugo's relURL drops the baseURL path component for inputs that already begin with "/". Already-resolved paths (anything a Hugo .RelPermalink produced, which already carries the baseURL subdirectory) pass through untouched. Without this check a "/study/locations/leeds-tutors/gcse/maths/" input came back out as "/study/study/locations/..." and 404'd. External URLs, anchors, mailto: and tel: are passed through untouched. */ -}} {{- $u := . -}} {{- $basePath := or (urls.Parse site.BaseURL).Path "/" -}} {{- if and (ne $basePath "/") (hasPrefix $u $basePath) -}} {{- $u -}} {{- else if hasPrefix $u "/" -}} {{- strings.TrimPrefix "/" $u | relURL -}} {{- else -}} {{- $u -}} {{- end -}}
